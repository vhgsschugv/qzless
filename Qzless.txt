local function MakeCheckWithKeybindAndInput(par, txt, y, checked, hasInput, defaultInputVal, callback, inputCallback, noKeybind)
    local btn = Instance.new("TextButton", par)
    btn.Size = UDim2.new(1, -16, 0, 24)
    btn.Position = UDim2.new(0, 8, 0, y)
    btn.BackgroundTransparency = 1
    btn.Text = ""
    local box = Instance.new("Frame", btn)
    box.Size = UDim2.new(0, 12, 0, 12)
    box.Position = UDim2.new(0, 0, 0.5, -6)
    box.BackgroundColor3 = checked and accentColor or bgColor
    box.BorderSizePixel = 1
    box.BorderColor3 = borderColor
    local lbl = Instance.new("TextLabel", btn)
    lbl.Size = UDim2.new(0, 0, 1, 0)
    lbl.Position = UDim2.new(0, 20, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = txt
    lbl.TextColor3 = textColor
    lbl.Font = Enum.Font.Code
    lbl.TextSize = 11
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.AutomaticSize = Enum.AutomaticSize.X

    local keyBtn = nil
    local inputBox = nil

    if not noKeybind then
        keyBtn = Instance.new("TextButton", btn)
        keyBtn.Size = UDim2.new(0, 32, 0, 18)
        keyBtn.Position = UDim2.new(0, 108, 0.5, -9)
        keyBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
        keyBtn.BorderColor3 = borderColor
        keyBtn.BorderSizePixel = 1
        keyBtn.Text = "..."
        keyBtn.TextColor3 = textColor
        keyBtn.Font = Enum.Font.Code
        keyBtn.TextSize = 10

        if hasInput then
            inputBox = Instance.new("TextBox", btn)
            inputBox.Size = UDim2.new(0, 40, 0, 18)
            inputBox.Position = UDim2.new(0, 144, 0.5, -9)
            inputBox.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
            inputBox.BorderColor3 = borderColor
            inputBox.BorderSizePixel = 1
            inputBox.Text = tostring(defaultInputVal)
            inputBox.TextColor3 = textColor
            inputBox.Font = Enum.Font.Code
            inputBox.TextSize = 10
            inputBox.FocusLost:Connect(function()
                local num = tonumber(inputBox.Text)
                if num and inputCallback then inputCallback(num) end
            end)
        end

        task.spawn(function()
            task.wait()
            if lbl and keyBtn then
                local offset = 20 + lbl.AbsoluteSize.X + 15 + 8
                keyBtn.Position = UDim2.new(0, offset, 0.5, -9)
                if inputBox then inputBox.Position = UDim2.new(0, offset + 36 + 8, 0.5, -9) end
            end
        end)
    end

    local boundKey = nil
    local waitingForKey = false
    local isChecked = checked

    if keyBtn then
        keyBtn.MouseButton1Click:Connect(function()
            waitingForKey = true
            keyBtn.Text = "..."
        end)
    end

    UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if waitingForKey then
            if input.UserInputType == Enum.UserInputType.Keyboard then
                boundKey = input.KeyCode
                keyBtn.Text = boundKey.Name
                waitingForKey = false
            end
        elseif boundKey and not gameProcessed and input.KeyCode == boundKey then
            isChecked = not isChecked
            box.BackgroundColor3 = isChecked and accentColor or bgColor
            if callback then callback(isChecked) end
        end
    end)

    btn.MouseButton1Click:Connect(function()
        if not waitingForKey then
            isChecked = not isChecked
            box.BackgroundColor3 = isChecked and accentColor or bgColor
            if callback then callback(isChecked) end
        end
    end)
end

    local function update(input)
        local rx = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        knob.Position = UDim2.new(rx, -3, 0.5, -2)

                local box = espGui:FindFirstChild(player.Name)

            end)
        end
        character.AncestryChanged:Connect(function()
            if not character:IsDescendantOf(workspace) then
                local box = espGui:FindFirstChild(player.Name)
                if box then box:Destroy() end
            end
        end)
    end

    p.CharacterAdded:Connect(OnCharacterAdded)

RunService.RenderStepped:Connect(function()
    if not EspEnabled then
        for _, v in pairs(espGui:GetChildren()) do
            if v:IsA("Frame") then v.Visible = false end
        end
        return
    end
    local finalColor = espColor
    if EspRainbow then
        local hue = (tick() % 2) / 2
        finalColor = Color3.fromHSV(hue, 1, 1)
    end
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local box = espGui:FindFirstChild(p.Name)
            if hum and hum.Health > 0 then
                local minX, maxX = math.huge, -math.huge
                local minY, maxY = math.huge, -math.huge
                local minZ, maxZ = math.huge, -math.huge
                local validParts = 0
                for _, part in pairs(p.Character:GetChildren()) do
                    if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" and part.Transparency < 1 then
                        local pos = part.Position
                        local size = part.Size / 2
                        minX = math.min(minX, pos.X - size.X)
                        maxX = math.max(maxX, pos.X + size.X)
                        minY = math.min(minY, pos.Y - size.Y)
                        maxY = math.max(maxY, pos.Y + size.Y)
                        minZ = math.min(minZ, pos.Z - size.Z)
                        maxZ = math.max(maxZ, pos.Z + size.Z)
                        validParts = validParts + 1
                    end
                end
                if validParts > 0 then
                    local corners = {
                        Vector3.new(minX, minY, minZ), Vector3.new(maxX, minY, minZ),
                        Vector3.new(minX, maxY, minZ), Vector3.new(maxX, maxY, minZ),
                        Vector3.new(minX, minY, maxZ), Vector3.new(maxX, minY, maxZ),
                        Vector3.new(minX, maxY, maxZ), Vector3.new(maxX, maxY, maxZ)
                    }
                    local minScreenX, maxScreenX = math.huge, -math.huge
                    local minScreenY, maxScreenY = math.huge, -math.huge
                    local allOnScreen = true
                    for _, corner in ipairs(corners) do
                        local screenPos, onScreen = Camera:WorldToViewportPoint(corner)
                        if onScreen and screenPos.Z > 0 then
                            if screenPos.X < minScreenX then minScreenX = screenPos.X end
                            if screenPos.X > maxScreenX then maxScreenX = screenPos.X end
                            if screenPos.Y < minScreenY then minScreenY = screenPos.Y end
                            if screenPos.Y > maxScreenY then maxScreenY = screenPos.Y end
                        else
                            allOnScreen = false
                        end
                    end
                    if allOnScreen and minScreenX < math.huge then
                        if not box then
                            box = Instance.new("Frame", espGui)
                            box.Name = p.Name
                            box.BackgroundTransparency = 1
                            box.BorderSizePixel = 0
                            local stroke = Instance.new("UIStroke", box)
                            stroke.Thickness = 1.5
                        end
                        box:FindFirstChildOfClass("UIStroke").Color = finalColor
                        local width = maxScreenX - minScreenX
                        local height = maxScreenY - minScreenY
                        local padding = 6
                        box.Size = UDim2.new(0, width + padding * 2, 0, height + padding * 2)
                        box.Position = UDim2.new(0, minScreenX - padding, 0, minScreenY - padding)
                        box.Visible = true
                    else
                        if box then box.Visible = false end
                    end
                end
            else
                if box then box.Visible = false end
            end
        end
    end
end)

local AuraEngine = {}
local auraState = {
    loaded = false, tree = nil, instances = {}, baseValues = {},
    roots = {}, rig = nil, currentName = nil, connections = {},
}

local BONE_MAP = {
    Head = { "Head" }, UpperTorso = { "UpperTorso", "Torso" },
    LowerTorso = { "LowerTorso" },
    RightUpperArm = { "RightUpperArm", "Right Arm" }, RightLowerArm = { "RightLowerArm", "Right Arm" },
    RightHand = { "RightHand", "Right Arm" },
    LeftUpperArm = { "LeftUpperArm", "Left Arm" }, LeftLowerArm = { "LeftLowerArm", "Left Arm" },
    LeftHand = { "LeftHand", "Left Arm" },
    RightUpperLeg = { "RightUpperLeg", "Right Leg" }, RightLowerLeg = { "RightLowerLeg", "Right Leg" },
    LeftUpperLeg = { "LeftUpperLeg", "Left Leg" }, LeftLowerLeg = { "LeftLowerLeg", "Left Leg" },
    HumanoidRootPart = { "HumanoidRootPart" }, Torso = { "Torso", "UpperTorso" },
}

local function newInstance(c)
    local ok, i = pcall(Instance.new, c)
    return ok and i or nil
end

local function applyProp(inst, key, value)
    local enumType
    if key == "Material" then enumType = Enum.Material
    elseif key == "Shape" and inst:IsA("BasePart") then enumType = Enum.PartType
    elseif key == "Shape" and inst:IsA("ParticleEmitter") then enumType = Enum.ParticleEmitterShape
    elseif key == "ShapeInOut" then enumType = Enum.ParticleEmitterShapeInOut
    elseif key == "ShapeStyle" then enumType = Enum.ParticleEmitterShapeStyle
    elseif key == "EmissionDirection" then enumType = Enum.NormalId
    elseif key == "Orientation" then enumType = Enum.ParticleOrientation
    end
    local resolved = value
    if type(value) == "string" then
        local short = value:match("^Enum%.[^.]+%.(.+)$")
        if short and enumType then resolved = enumType[short] or value end
    end
    pcall(function() inst[key] = resolved end)
end

local function resolveBone(node)
    for _, name in ipairs(BONE_MAP[node.name] or {}) do
        local f = auraState.rig:FindFirstChild(name)
        if f and f:IsA("BasePart") then return f end
    end
    return nil
end

function AuraEngine.Load(tree, character)
    AuraEngine.Unload()
    if type(tree) ~= "table" then return false end
    auraState.tree = tree
    auraState.rig = character or (LP and LP.Character)
    if not auraState.rig then return false end
    local nodes = tree.nodes or {}
    local byId = {}
    for _, n in ipairs(nodes) do byId[n.id] = n end
    for _, n in ipairs(nodes) do
        local inst = newInstance(n.class)
        if not inst then continue end
        for k, v in pairs(n.props or {}) do
            if k ~= "CFrame" then applyProp(inst, k, v) end
        end
        auraState.instances[n.id] = inst
    end
    for _, n in ipairs(nodes) do
        local inst = auraState.instances[n.id]
        if not inst then continue end
        if n.parent == 0 then
            local bone = resolveBone(n)
            if bone then
                inst.Name = "AuraRoot_" .. n.name
                inst.CanCollide, inst.CanTouch, inst.CanQuery = false, false, false
                inst.Massless, inst.CastShadow, inst.Transparency = true, false, 1
                inst.CFrame = bone.CFrame
                inst.Parent = bone
                local w = Instance.new("WeldConstraint")
                w.Part0, w.Part1, w.Parent = inst, bone, inst
                auraState.roots[n.id] = inst
            end
        else
            local p = auraState.instances[n.parent]
            if p then inst.Parent = p end
        end
    end
    for _, link in ipairs(tree.links or {}) do
        local beam, a0, a1 = auraState.instances[link[1]], auraState.instances[link[2]], auraState.instances[link[3]]
        if beam and a0 and a1 then
            pcall(function() beam.Attachment0 = a0; beam.Attachment1 = a1 end)
        end
    end
    for _, n in ipairs(nodes) do
        local inst = auraState.instances[n.id]
        if inst and inst:IsA("Attachment") and n.props and n.props.CFrame then
            pcall(function() inst.CFrame = n.props.CFrame end)
        end
    end
    auraState.loaded = true
    auraState.currentName = tree.name
    print(("[aura] loaded: %s"):format(tostring(tree.name)))
    return true
end

function AuraEngine.Unload()
    for _, inst in pairs(auraState.instances) do
        if inst and inst.Parent then pcall(function() inst:Destroy() end) end
    end
    auraState.instances, auraState.roots, auraState.baseValues = {}, {}, {}
    auraState.loaded, auraState.tree, auraState.currentName = false, nil, nil
end

local CYAN = Color3.fromRGB(133, 220, 255)
local CYAN_SEQ = ColorSequence.new({
    ColorSequenceKeypoint.new(0, CYAN),
    ColorSequenceKeypoint.new(1, CYAN),
})

-- NumberSequence 快速建立：ns({t, v[, envelope]}, ...)
local function ns(...)
    local keys = {}
    for _, k in ipairs({...}) do
        table.insert(keys, NumberSequenceKeypoint.new(k[1], k[2], k[3] or 0))
    end
    return NumberSequence.new(keys)
end

local function PartNode(id, name, size, cf, canCollide, color)
    return { id = id, parent = 0, class = "Part", name = name, props = {
        Size = size, Color = color or Color3.fromRGB(127, 127, 127),
        Material = Enum.Material.Plastic, Transparency = 1, Reflectance = 0,
        CastShadow = true, Shape = Enum.PartType.Block, Anchored = false,
        CanCollide = canCollide, CanQuery = true, CanTouch = true, Massless = false,
        CFrame = cf,
    } }
end

local function AttNode(id, parent, name, cf)
    return { id = id, parent = parent, class = "Attachment", name = name, props = {
        CFrame = cf, Visible = false,
    } }
end

local AntiHitEnabled = false
local antiHitConn = nil
local AuraEnabled = false

local function startAntiHit()
    if antiHitConn then return end
    antiHitConn = RunService.Heartbeat:Connect(function(dt)
        if not AntiHitEnabled then return end
        local char = LP.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then return end
        hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        if hum.MoveDirection.Magnitude < 0.1 and hum.FloorMaterial ~= Enum.Material.Air then
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
        hrp.AssemblyAngularVelocity = Vector3.zero
        hrp.RotVelocity = Vector3.zero
        local pos = hrp.Position
        local rot = CFrame.Angles(math.rad(180), tick() * 30, 0)
        hrp.CFrame = CFrame.new(pos.X, pos.Y, pos.Z) * rot
    end)
end
local function stopAntiHit()
    AntiHitEnabled = false
    if antiHitConn then antiHitConn:Disconnect(); antiHitConn = nil end
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true) end
end

MakeCheckWithKeybindAndInput(gPlayerMod, "Anti-Hit", 18, false, false, 0, function(state)
    AntiHitEnabled = state
    if state then startAntiHit() else stopAntiHit() end
end, nil, true)

    end
end, nil, true)
